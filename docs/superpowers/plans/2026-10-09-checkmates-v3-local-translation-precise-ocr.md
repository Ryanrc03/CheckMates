# CheckMates V3 Local Translation + Precise OCR Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship two user-facing features on branch `feat/v3-local-translation-precise-ocr`: (1) a `Show Chinese names` toggle that translates dish names with rules + glossary + the on-device Firefox Translations (Bergamot) en→zh-Hans model, and (2) a `Precise` recognition engine (PaddleOCR PP-OCRv5 mobile in a Worker) selectable on the photo-adjustment page; both share one local model-asset layer (manifest, Cache API, progress, expiry, removal).

**Architecture:** A `src/lib/model-assets/` layer downloads versioned files from same-origin `/models/<bundle>/<version>/…` (generated by `scripts/prepare-model-assets.mjs`) into Cache API buckets `cm-models-<id>-v<version>`. Translation = pure text pipeline (`src/lib/translation/`) + a classic Worker running Mozilla's bergamot build; results live in a zustand store and render through `DishLabel`. Precise OCR = module Worker bundled by Turbopack that imports `@paddleocr/paddleocr-js` in non-worker mode, returns text boxes, which `spatialRows` turns into lines for the unchanged `parseReceiptText`.

**Tech Stack:** Next.js 16.3.6 (Turbopack), React 19, TypeScript strict, zustand 5, Vitest 5, Playwright 1.58.2 (Chromium 145), `@paddleocr/paddleocr-js` 0.4.2 + `onnxruntime-web` 1.24.3 (exact), Mozilla bergamot-translator v0.6.0 wasm (vendored glue, MPL-2.0).

**Spec:** [design](../specs/2026-10-09-checkmates-v3-local-translation-precise-ocr-design.md). **Delivery boundary:** develop + local tests only; no push, no deploy; true-device validation is out of scope and every doc must say "desktop verified only".

```js
/* CheckMates translation worker. Runs Mozilla's bergamot-translator v0.6.0 (MPL-2.0) build.
 * Protocol: main thread posts { id, name, args } and receives { id, result } or { id, error: { name, message } }.
 * Methods: initialize({ wasm: ArrayBuffer }) -> { version }
 *          loadModel({ model, lex, vocabs: [src, trg], from, to }) -> { loaded: true }
 *          translate({ texts: string[] }) -> string[]
 *          free() -> { freed: true }
 */
importScripts("./bergamot-translator.js");

var ALIGN = { model: 256, lex: 64, vocab: 64 };
var bergamot = null;
var service = null;
var model = null;

function configText(config) {
  var indent = "            ";
  var out = "\n";
  Object.keys(config).forEach(function (key) { out += indent + key + ": " + config[key] + "\n"; });
  return out + indent;
}
function aligned(buffer, alignment) {
  var memory = new bergamot.AlignedMemory(buffer.byteLength, alignment);
  memory.getByteArrayView().set(new Uint8Array(buffer));
  return memory;
}
var handlers = {
  initialize: function (args) {
    return new Promise(function (resolve, reject) {
      var M = loadBergamot({
        INITIAL_MEMORY: 41943040,
        print: function () {},
        printErr: function (line) { console.warn("[bergamot]", line); },
        onAbort: function (reason) { reject(new Error("bergamot wasm aborted: " + reason)); },
        onRuntimeInitialized: function () { Promise.resolve().then(function () { bergamot = M; service = new bergamot.BlockingService({ cacheSize: 0 }); resolve({ version: "v0.6.0" }); }); },
        wasmBinary: args.wasm,
      });
    });
  },
  loadModel: function (args) {
    if (!bergamot) throw new Error("initialize first");
    if (model) { model.delete(); model = null; }
    var memory = aligned(args.model, ALIGN.model);
    var lex = args.lex ? aligned(args.lex, ALIGN.lex) : null;
    var vocabs = new bergamot.AlignedMemoryList();
    args.vocabs.forEach(function (v) { vocabs.push_back(aligned(v, ALIGN.vocab)); });
    var config = configText({ "beam-size": "1", normalize: "1.0", "word-penalty": "0", "max-length-break": "128", "mini-batch-words": "1024", workspace: "128", "max-length-factor": "2.0", "skip-cost": "true", "cpu-threads": "0", quiet: "true", "quiet-translation": "true", "gemm-precision": "int8shiftAlphaAll", alignment: "soft" });
    model = new bergamot.TranslationModel(args.from || "en", args.to || "zh", config, memory, lex, vocabs, null);
    return { loaded: true };
  },
  translate: function (args) {
    if (!model) throw new Error("loadModel first");
    var messages = new bergamot.VectorString();
    var options = new bergamot.VectorResponseOptions();
    args.texts.forEach(function (text) { messages.push_back(text); options.push_back({ qualityScores: false, alignment: false, html: false }); });
    var responses = null;
    try {
      responses = service.translate(model, messages, options);
      return args.texts.map(function (_, i) { return responses.get(i).getTranslatedText(); });
    } finally { messages.delete(); options.delete(); if (responses) responses.delete(); }
  },
  free: function () { if (model) { model.delete(); model = null; } return { freed: true }; },
};
self.onmessage = function (event) {
  var data = event.data || {};
  Promise.resolve().then(function () {
    if (typeof handlers[data.name] !== "function") throw new TypeError("Unknown method " + data.name);
    return handlers[data.name](data.args || {});
  }).then(function (result) { self.postMessage({ id: data.id, result: result }); }, function (error) { self.postMessage({ id: data.id, error: { name: error && error.name, message: String(error && error.message || error) } }); });
};
```

// src/lib/model-assets/bundles.ts
import { bundleUrl, loadManifest } from "./manifest";
import { readRecords, writeRecords } from "./records";
import type { AssetEnv, BundleHandle, BundleProgress, BundleStatus, ModelBundleId, ModelBundleManifest } from "./types";
export const CACHE_PREFIX = "cm-models-";
export const cacheNameFor = (id: ModelBundleId, version: string) => `${CACHE_PREFIX}${id}-v${version}`;
export function browserEnv(): AssetEnv { return { caches: globalThis.caches, storage: window.localStorage, fetch: (input, init) => fetch(input, init), origin: window.location.origin, now: () => Date.now() }; }
const abortError = () => new DOMException("Download canceled.", "AbortError");
async function sha256Hex(buffer: ArrayBuffer): Promise<string> { const digest = await crypto.subtle.digest("SHA-256", buffer); return [...new Uint8Array(digest)].map(b => b.toString(16).padStart(2, "0")).join(""); }
async function bundleCacheNames(env: AssetEnv, id: ModelBundleId): Promise<string[]> { return (await env.caches.keys()).filter(name => name.startsWith(`${CACHE_PREFIX}${id}-v`)); }
async function cachedFiles(env: AssetEnv, bundle: ModelBundleManifest): Promise<Set<string>> {
  const cache = await env.caches.open(cacheNameFor(bundle.id, bundle.version)); const present = new Set<string>();
  for (const file of bundle.files) if (file.cache === "cache-api" && await cache.match(bundleUrl(env.origin, bundle, file.path))) present.add(file.path);
  return present;
}
export async function getBundleStatus(id: ModelBundleId, env: AssetEnv = browserEnv()): Promise<BundleStatus> {
  const bundle = (await loadManifest(env.fetch, env.origin)).bundles[id]; const present = await cachedFiles(env, bundle); const record = readRecords(env.storage).bundles[id];
  const cached = bundle.files.filter(f => f.cache === "cache-api").every(f => present.has(f.path));
  return { id, version: bundle.version, totalBytes: bundle.totalBytes, downloadBytes: bundle.downloadBytes, cached, installedAt: record?.installedAt ?? null, lastUsedAt: record?.lastUsedAt ?? null };
}
async function download(env: AssetEnv, url: string, expectedBytes: number, expectedSha: string, signal: AbortSignal | undefined, onChunk: (received: number) => void): Promise<Response> {
  if (signal?.aborted) throw abortError();
  const response = await env.fetch(url, { signal, cache: "no-store" });
  if (!response.ok || !response.body) throw new Error(`Could not download ${url.split("/").pop()} (HTTP ${response.status}).`);
  const reader = response.body.getReader(); const chunks: Uint8Array[] = []; let received = 0;
  for (;;) { const { done, value } = await reader.read(); if (done) break; chunks.push(value); received += value.byteLength; onChunk(received); }
  if (received !== expectedBytes) throw new Error(`Downloaded ${url.split("/").pop()} is ${received} bytes; expected ${expectedBytes}.`);
  const blob = new Blob(chunks); if (await sha256Hex(await blob.arrayBuffer()) !== expectedSha) throw new Error(`Downloaded ${url.split("/").pop()} failed its integrity check.`);
  return new Response(blob, { status: 200, headers: { "Content-Type": response.headers.get("Content-Type") ?? "application/octet-stream", "Content-Length": String(received) } });
}
function touchRecord(env: AssetEnv, id: ModelBundleId, version: string): void {
  const records = readRecords(env.storage); const existing = records.bundles[id]; const now = env.now();
  records.bundles[id] = { version, installedAt: existing && existing.version === version ? existing.installedAt : now, lastUsedAt: now }; writeRecords(env.storage, records);
}
export async function ensureBundle(id: ModelBundleId, { signal, onProgress, env = browserEnv() }: { signal?: AbortSignal; onProgress?: (progress: BundleProgress) => void; env?: AssetEnv } = {}): Promise<BundleHandle> {
  const bundle = (await loadManifest(env.fetch, env.origin)).bundles[id]; const name = cacheNameFor(id, bundle.version); const cache = await env.caches.open(name);
  const files = bundle.files.filter(f => f.cache === "cache-api"); const totalBytes = files.reduce((s, f) => s + f.bytes, 0); let loaded = 0;
  onProgress?.({ stage: "checking", loadedBytes: 0, totalBytes });
  for (const file of files) {
    const url = bundleUrl(env.origin, bundle, file.path);
    if (await cache.match(url)) { loaded += file.bytes; onProgress?.({ stage: "downloading", loadedBytes: loaded, totalBytes, file: file.path }); continue; }
    const base = loaded; const response = await download(env, url, file.bytes, file.sha256, signal, received => onProgress?.({ stage: "downloading", loadedBytes: base + received, totalBytes, file: file.path }));
    await cache.put(url, response); loaded += file.bytes;
  }
  for (const stale of await bundleCacheNames(env, id)) if (stale !== name) await env.caches.delete(stale);
  touchRecord(env, id, bundle.version); onProgress?.({ stage: "ready", loadedBytes: totalBytes, totalBytes });
  const url = (path: string) => bundleUrl(env.origin, bundle, path);
  return { id, version: bundle.version, cacheName: name, url,
    bytes: async path => { const hit = await cache.match(url(path)); if (!hit) throw new Error(`${path} is not in the local model cache.`); return hit.arrayBuffer(); },
    fetch: async (input, init) => { const target = typeof input === "string" ? input : input instanceof URL ? input.href : input.url; const hit = await cache.match(target); return hit ?? env.fetch(input, init); },
    touch: () => touchRecord(env, id, bundle.version) };
}
export async function removeBundle(id: ModelBundleId, env: AssetEnv = browserEnv()): Promise<void> {
  for (const name of await bundleCacheNames(env, id)) await env.caches.delete(name);
  const records = readRecords(env.storage); delete records.bundles[id]; writeRecords(env.storage, records);
}
export async function purgeExpiredBundles({ env = browserEnv(), maxAgeDays = 14 }: { env?: AssetEnv; maxAgeDays?: number } = {}): Promise<ModelBundleId[]> {
  const manifest = await loadManifest(env.fetch, env.origin); const records = readRecords(env.storage); const purged: ModelBundleId[] = []; const limit = maxAgeDays * 24 * 60 * 60 * 1000;
  for (const id of Object.keys(manifest.bundles) as ModelBundleId[]) {
    const record = records.bundles[id]; const names = await bundleCacheNames(env, id); if (!names.length && !record) continue;
    const expired = !record || record.version !== manifest.bundles[id].version || env.now() - record.lastUsedAt > limit;
    if (expired) { await removeBundle(id, env); purged.push(id); }
  }
  return purged;
}

// src/lib/translation/pipeline.ts
import { expandAbbreviations } from "./abbreviations";
import { lookupGlossary } from "./glossary";
import { normalizeSource } from "./normalize";
import { joinProtected, splitProtected, type ProtectedParts } from "./protect";
import { matchModifier } from "./rules";
import { MAX_SOURCE_LENGTH, type TranslationProvider } from "./types";
export type PhraseTranslation = { text: string; provider: TranslationProvider; expandedSource?: string };
export type ModelTranslate = (text: string) => Promise<string | null>;
export type TranslationPlan = { kind: "done"; result: PhraseTranslation } | { kind: "model"; modelInput: string; compose: (modelOutput: string) => PhraseTranslation } | { kind: "none" };
const CJK = /[㐀-鿿]/;
export function sanitizeModelOutput(output: string | null | undefined, input: string): string | null {
  if (!output) return null; const text = output.replace(/^["'“”‘’\s]+|["'“”‘’\s]+$/g, "").trim();
  if (!text || text.length > 200 || /[<>\u0000-\u001f]/.test(text) || !CJK.test(text) || text.toLowerCase() === input.toLowerCase()) return null;
  return text;
}
function finish(parts: ProtectedParts, text: string, provider: TranslationProvider, expandedSource: string | undefined): PhraseTranslation { return { text: joinProtected(parts, text), provider, expandedSource }; }
export function planTranslation(source: string): TranslationPlan {
  const normalized = normalizeSource(source); if (!normalized || normalized.length > MAX_SOURCE_LENGTH) return { kind: "none" };
  const parts = splitProtected(normalized); if (!parts.core || !/[a-z]/i.test(parts.core)) return { kind: "none" };
  const { text: core, expanded } = expandAbbreviations(parts.core); const expandedSource = expanded ? core : undefined;
  const direct = lookupGlossary(core); if (direct) return { kind: "done", result: finish(parts, direct, "glossary", expandedSource) };
  const modifier = matchModifier(core);
  if (modifier) {
    const rest = lookupGlossary(modifier.rest); if (rest) return { kind: "done", result: finish(parts, `${modifier.zhPrefix}${rest}`, "rule", expandedSource) };
    return { kind: "model", modelInput: modifier.rest, compose: output => finish(parts, `${modifier.zhPrefix}${output}`, "model", expandedSource) };
  }
  return { kind: "model", modelInput: core, compose: output => finish(parts, output, "model", expandedSource) };
}
export async function translatePhrase(source: string, model: ModelTranslate | null): Promise<PhraseTranslation | null> {
  const plan = planTranslation(source); if (plan.kind === "done") return plan.result; if (plan.kind === "none" || !model) return null;
  const output = sanitizeModelOutput(await model(plan.modelInput), plan.modelInput); return output ? plan.compose(output) : null;
}

import type { StorageLike } from "@/lib/model-assets/types";
import type { TranslationProvider } from "./types";
export const TRANSLATION_CACHE_KEY = "bitesplit-translations";
export const MAX_CACHE_ENTRIES = 300;
export const CACHE_TTL_MS = 30 * 24 * 60 * 60 * 1000;
export type CacheEntry = { text: string; provider: TranslationProvider; revision: string; at: number };
export type TranslationCache = { get(key: string, revision: string): CacheEntry | null; set(key: string, entry: CacheEntry): void; clear(): void };
type Stored = { version: 1; entries: Record<string, CacheEntry> };
export function createTranslationCache(storage: StorageLike | null, now: () => number = () => Date.now()): TranslationCache {
  let entries: Record<string, CacheEntry> | null = null;
  const load = (): Record<string, CacheEntry> => {
    if (entries) return entries; entries = {};
    try { const raw = storage?.getItem(TRANSLATION_CACHE_KEY); const parsed: unknown = raw ? JSON.parse(raw) : null; const stored = parsed as Stored | null; if (stored && stored.version === 1 && stored.entries && typeof stored.entries === "object") entries = { ...stored.entries }; } catch { entries = {}; }
    return entries;
  };
  const save = () => { try { storage?.setItem(TRANSLATION_CACHE_KEY, JSON.stringify({ version: 1, entries: load() } satisfies Stored)); } catch { /* keep the in-memory copy */ } };
  return {
    get(key, revision) { const entry = load()[key]; if (!entry || entry.revision !== revision || now() - entry.at > CACHE_TTL_MS) return null; entry.at = now(); return entry; },
    set(key, entry) { const all = load(); all[key] = entry; const keys = Object.keys(all); if (keys.length > MAX_CACHE_ENTRIES) { keys.sort((a, b) => all[a].at - all[b].at).slice(0, keys.length - MAX_CACHE_ENTRIES).forEach(k => delete all[k]); } save(); },
    clear() { entries = {}; try { storage?.removeItem(TRANSLATION_CACHE_KEY); } catch { /* ignore */ } },
  };
}

// src/store/useTranslationStore.ts
import { create } from "zustand";
import type { StorageLike } from "@/lib/model-assets/types";
import { getTranslationService } from "@/lib/translation/service";
import { translationKey } from "@/lib/translation/normalize";
import type { TranslatedItem, TranslationItem, TranslationProgress, TranslationService } from "@/lib/translation/types";
export const TRANSLATION_ENABLED_KEY = "bitesplit-translation-enabled";
export type TranslationStoreState = {
  enabled: boolean; results: Record<string, TranslatedItem>; pending: Record<string, true>; progress: TranslationProgress | null; busy: boolean; error: string | null; needsDownload: boolean; downloadBytes: number; modelAllowed: boolean; modelUnavailableReason: string | null;
  setEnabled: (enabled: boolean) => void; request: (items: TranslationItem[]) => Promise<void>; allowModelAndRequest: (items: TranslationItem[]) => Promise<void>; cancel: () => void; clearForBill: () => void; invalidate: (id: string) => void;
};
export function createTranslationStore(service: TranslationService, storage: StorageLike | null) {
  let generation = 0; let controller: AbortController | null = null;
  const read = () => { try { return storage?.getItem(TRANSLATION_ENABLED_KEY) === "1"; } catch { return false; } };
  return create<TranslationStoreState>((set, get) => {
    const run = async (items: TranslationItem[], allowModel: boolean) => {
      if (!get().enabled) return;
      const id = ++generation; controller?.abort(); controller = new AbortController(); const signal = controller.signal;
      const current = new Map(items.map(i => [i.id, translationKey(i.name)]));
      const kept = Object.fromEntries(Object.entries(get().results).filter(([key, r]) => current.get(key) === translationKey(r.sourceText)));
      const pending = Object.fromEntries(items.filter(i => !kept[i.id] && translationKey(i.name)).map(i => [i.id, true as const]));
      set({ results: kept, pending, error: null, busy: true, needsDownload: false });
      try {
        const capabilities = await service.getCapabilities(); if (id !== generation) return;
        const useModel = allowModel && capabilities.modelAvailable;
        set({ downloadBytes: capabilities.downloadBytes, modelUnavailableReason: capabilities.modelAvailable ? null : capabilities.reason ?? "unavailable" });
        const outcome = await service.translateDishes(items.filter(i => !kept[i.id]), { signal, allowModel: useModel, onProgress: progress => { if (id === generation) set({ progress }); },
          onItem: item => { if (id !== generation || current.get(item.id) !== translationKey(item.sourceText)) return; set(state => { const next = { ...state.pending }; delete next[item.id]; return { results: { ...state.results, [item.id]: item }, pending: next }; }); } });
        if (id !== generation) return;
        const unresolved = items.some(i => translationKey(i.name) && !get().results[i.id]);
        set({ pending: {}, progress: null, busy: false, error: outcome.error, needsDownload: unresolved && !allowModel && capabilities.modelAvailable && !outcome.error });
      } catch (error) { if (id !== generation) return; set({ pending: {}, progress: null, busy: false, error: error instanceof DOMException && error.name === "AbortError" ? null : error instanceof Error ? error.message : "Translation failed." }); }
    };
    return {
      enabled: read(), results: {}, pending: {}, progress: null, busy: false, error: null, needsDownload: false, downloadBytes: 0, modelAllowed: false, modelUnavailableReason: null,
      setEnabled: enabled => { try { storage?.setItem(TRANSLATION_ENABLED_KEY, enabled ? "1" : "0"); } catch { /* ignore */ } set({ enabled }); if (!enabled) get().cancel(); },
      request: items => run(items, get().modelAllowed),
      allowModelAndRequest: async items => { set({ modelAllowed: true, needsDownload: false }); await run(items, true); },
      cancel: () => { generation++; controller?.abort(); controller = null; void service.cancelTranslations(); set({ pending: {}, progress: null, busy: false }); },
      clearForBill: () => { generation++; controller?.abort(); controller = null; set({ results: {}, pending: {}, progress: null, busy: false, error: null, needsDownload: false }); },
      invalidate: id => set(state => { const results = { ...state.results }; delete results[id]; return { results }; }),
    };
  });
}

// src/lib/ocr/paddle/spatial.ts
export type OcrBox = { text: string; score?: number; poly?: [number, number][]; bbox?: { x0: number; y0: number; x1: number; y1: number } };
export type SpatialLine = { text: string; bbox: { x0: number; y0: number; x1: number; y1: number }; confidence: number };
export type SpatialRows = { lines: SpatialLine[]; text: string; estimatedAngleDeg: number; meanScore: number };
const corners = (box: OcrBox): [number, number][] => box.poly && box.poly.length >= 4 ? box.poly : box.bbox ? [[box.bbox.x0, box.bbox.y0], [box.bbox.x1, box.bbox.y0], [box.bbox.x1, box.bbox.y1], [box.bbox.x0, box.bbox.y1]] : [];
export function spatialRows(boxes: OcrBox[]): SpatialRows {
  const usable = boxes.filter(b => b.text.trim() && corners(b).length >= 4); if (!usable.length) return { lines: [], text: "", estimatedAngleDeg: 0, meanScore: 0 };
  const angles = usable.filter(b => b.poly && /[a-z]/i.test(b.text)).map(b => { const [p0, p1] = b.poly!; return { angle: Math.atan2(p1[1] - p0[1], p1[0] - p0[0]), width: Math.hypot(p1[0] - p0[0], p1[1] - p0[1]), height: Math.hypot(b.poly![3][0] - p0[0], b.poly![3][1] - p0[1]) }; }).filter(a => a.width > 3 * a.height).map(a => a.angle).sort((a, b) => a - b);
  const angle = angles.length ? angles[Math.floor(angles.length / 2)] : 0; const co = Math.cos(angle); const si = Math.sin(angle);
  const transformed = usable.map(b => { const pts = corners(b).map(([x, y]) => [x * co + y * si, -x * si + y * co] as [number, number]); const xs = pts.map(p => p[0]); const ys = pts.map(p => p[1]); const bbox = { x0: Math.min(...xs), y0: Math.min(...ys), x1: Math.max(...xs), y1: Math.max(...ys) }; return { box: b, bbox, cy: (bbox.y0 + bbox.y1) / 2, height: bbox.y1 - bbox.y0 }; }).sort((a, b) => a.cy - b.cy);
  const groups: { cy: number; height: number; members: typeof transformed }[] = [];
  for (const item of transformed) {

// Runs the whole PaddleOCR SDK (OpenCV.js + ONNX Runtime) inside this worker so nothing heavy touches the main thread.
import { PaddleOCR } from "@paddleocr/paddleocr-js";
import * as ort from "onnxruntime-web";
import { PADDLE_MODELS, type PaddleInitPayload, type PaddleRequest, type PaddleResponse } from "./protocol";
type MatLike = { delete(): void }; type CvLike = { matFromImageData(imageData: ImageData): MatLike };
type SourceMat = { width: number; height: number; mat: MatLike; dispose(): void };
type Engine = Awaited<ReturnType<typeof PaddleOCR.create>>;

// The SDK's built-in sourceToMat calls document.createElement("canvas") and cv.imread references HTMLImageElement (ReferenceError in workers).
async function workerSourceToMat(cv: CvLike, source: unknown): Promise<SourceMat> {
  const bitmap = source instanceof ImageBitmap ? source : await createImageBitmap(source as Blob);
  const canvas = new OffscreenCanvas(bitmap.width, bitmap.height); const ctx = canvas.getContext("2d", { willReadFrequently: true });
  if (!ctx) throw new Error("OffscreenCanvas 2D context unavailable in worker."); ctx.drawImage(bitmap, 0, 0);
  const mat = cv.matFromImageData(ctx.getImageData(0, 0, bitmap.width, bitmap.height));
  return { width: bitmap.width, height: bitmap.height, mat, dispose() { mat.delete(); bitmap.close(); } };
}
// ort.env.logLevel does not silence ORT's 27 "CleanUnusedInitializersAndNodeArgs" console.error lines per session; the per-session logSeverityLevel does.
function quietOrt() {
  const factory = ort.InferenceSession as unknown as { create: (...args: unknown[]) => Promise<unknown>; __checkmatesPatched?: boolean };
  if (factory.__checkmatesPatched) return; const original = factory.create.bind(ort.InferenceSession);
  factory.create = (model: unknown, options?: unknown) => original(model, { logSeverityLevel: 3, ...(options as object) }); factory.__checkmatesPatched = true; ort.env.logLevel = "error";
}
function cacheBackedFetch(cacheName: string): typeof fetch {
  return async (input, init) => { const url = typeof input === "string" ? input : input instanceof URL ? input.href : input.url; try { const hit = await (await caches.open(cacheName)).match(url); if (hit) return hit; } catch { /* fall through */ } return fetch(input, init); };
}
async function init(payload: PaddleInitPayload) {
  quietOrt();
  const instance = await PaddleOCR.create({ worker: false, initialize: false, fetch: cacheBackedFetch(payload.cacheName), textDetectionModelName: PADDLE_MODELS.det, textRecognitionModelName: PADDLE_MODELS.rec, textDetectionModelAsset: { url: payload.detUrl }, textRecognitionModelAsset: { url: payload.recUrl }, ortOptions: { backend: "wasm", wasmPaths: payload.wasmPaths, numThreads: 1, proxy: false } });
  (instance as unknown as { sourceToMat: unknown }).sourceToMat = workerSourceToMat; // OcrPipelineRunner stores it as a plain instance field
  const summary = await instance.initialize(); engine = instance; return { backend: summary.backend, elapsedMs: summary.elapsedMs };
}
async function predict(blob: Blob) {
  if (!engine) throw new Error("Engine not initialized."); const [result] = await engine.predict(blob);
  return { items: result.items.map(i => ({ text: i.text, score: i.score, poly: i.poly.map(p => [p[0], p[1]] as [number, number]) })), metrics: result.metrics };
}
async function dispose() { if (engine) { await engine.dispose(); engine = null; } return true; }
self.onmessage = async (event: MessageEvent<PaddleRequest>) => {
  const message = event.data; const reply = (response: PaddleResponse) => self.postMessage(response);
  try {
    if (message.type === "init") reply({ id: message.id, ok: true, result: await init(message.payload) });
    else if (message.type === "predict") reply({ id: message.id, ok: true, result: await predict(message.payload) });
    else reply({ id: message.id, ok: true, result: await dispose() });
  } catch (error) { reply({ id: message.id, ok: false, error: error instanceof Error ? error.message : String(error) }); }
};

type ProgressMessage = { status: string; progress?: number };
export type OcrWorker = { recognize: (image: Blob) => Promise<{ data: { text: string; confidence: number } }>; terminate: () => Promise<unknown> };
export type OcrEngine = "standard" | "precise";
type Options = { signal: AbortSignal; onProgress: (stage: string, progress: number | null) => void; engine?: OcrEngine };
type Factory = (logger: (message: ProgressMessage) => void, signal: AbortSignal) => Promise<OcrWorker>;
const aborted = () => new DOMException("Recognition canceled. You can select another photo or enter items manually.", "AbortError");

export function createReceiptRecognizer(factories: Record<OcrEngine, Factory>) {
  let cancelActive: (() => void) | null = null;
  return async (image: Blob, { signal, onProgress, engine = "standard" }: Options): Promise<{ text: string; confidence: number }> => {
    cancelActive?.();
    if (signal.aborted) throw aborted();
    let ended = false; let worker: OcrWorker | null = null; let terminated = false;
    const release = () => { if (worker && !terminated) { terminated = true; void worker.terminate().catch(() => {}); } };
    let rejectStop!: (error: Error) => void;
    const stop = new Promise<never>((_, reject) => { rejectStop = reject; });
    const cancel = () => { ended = true; release(); rejectStop(aborted()); };
    cancelActive = cancel; signal.addEventListener("abort", cancel, { once: true });
    const timer = setTimeout(() => { ended = true; release(); rejectStop(new Error("Recognition exceeded 120 seconds. Try a smaller, clearer photo or enter items manually.")); }, 120000);
    const work = (async () => {
      onProgress(engine === "precise" ? "Preparing precise recognition" : "Loading recognition resources", null);
      worker = await factories[engine](message => { if (!ended) onProgress(message.status, typeof message.progress === "number" && Number.isFinite(message.progress) ? Math.max(0, Math.min(1, message.progress)) : null); }, signal);

      const { data } = await worker.recognize(image);
      if (ended) throw aborted();
      if (!data.text.trim()) throw new Error("No text found. Try a clearer receipt photo or enter items manually.");
      return { text: data.text, confidence: data.confidence };
    })();
    try { return await Promise.race([work, stop]); }
    finally { ended = true; clearTimeout(timer); signal.removeEventListener("abort", cancel); release(); if (cancelActive === cancel) cancelActive = null; }
  };
}

// src/components/home/offline-data.tsx
import { useEffect, useState } from "react";
import { Button } from "@/components/ui/button";
import { getBundleStatus, removeBundle } from "@/lib/model-assets/bundles";
import type { BundleStatus, ModelBundleId } from "@/lib/model-assets/types";
import { createTranslationCache } from "@/lib/translation/cache";
import { useTranslationStore } from "@/store/useTranslationStore";
const LABELS: Record<ModelBundleId, string> = { "translate-enzh": "translation model", "ocr-precise": "precise recognition model" };
const mb = (bytes: number) => `${Math.max(1, Math.round(bytes / 1_000_000))} MB`;
export function OfflineData() {
  const [rows, setRows] = useState<BundleStatus[] | null>(null); const [error, setError] = useState<string | null>(null);
  const refresh = async () => { try { setRows(await Promise.all((["translate-enzh", "ocr-precise"] as ModelBundleId[]).map(id => getBundleStatus(id)))); setError(null); } catch { setError("Offline data status is unavailable right now."); setRows([]); } };
  useEffect(() => { void refresh(); }, []);
  return <details className="offline-data paper"><summary>Offline data</summary>
    <p className="muted">Models are stored in this browser only and removed automatically after 14 days without use. iOS may also remove them after 7 days without use.</p>
    {error && <p className="notice">{error}</p>}
    {rows?.map(row => <div className="offline-row" key={row.id}><span>{LABELS[row.id]} · {row.cached ? `Downloaded · ${mb(row.totalBytes)}` : "Not downloaded"}</span>
      {row.cached && <Button variant="outline" aria-label={`Remove ${LABELS[row.id]}`} onClick={async () => { await removeBundle(row.id); if (row.id === "translate-enzh") { createTranslationCache(window.localStorage).clear(); useTranslationStore.getState().clearForBill(); } await refresh(); }}>Remove</Button>}</div>)}
  </details>;
}
